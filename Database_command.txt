CREATE TABLE users (
    id INT NOT NULL AUTO_INCREMENT,
    username VARCHAR(25) NOT NULL,
    email VARCHAR(255) NOT NULL,
    password VARCHAR(255) NOT NULL,
    img VARCHAR(255) DEFAULT NULL, -- Allow NULL values for img
    PRIMARY KEY (id)
);


CREATE TABLE posts (
    id INT NOT NULL AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    des VARCHAR(1000) NOT NULL,
    img VARCHAR(255) DEFAULT NULL, -- Allows NULL in case of missing image
    date DATETIME NOT NULL,
    uid INT NOT NULL,
    cat VARCHAR(50), -- New column to store the category if required
    PRIMARY KEY (id),
    INDEX uid_idx (uid ASC),
    CONSTRAINT fk_uid
        FOREIGN KEY (uid)
        REFERENCES blog.users(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

-- Table to store comments on posts
CREATE TABLE comments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    post_id INT NOT NULL,  -- Foreign key for posts
    user_id INT NOT NULL,  -- Assuming you have a users table
    comment TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (post_id) REFERENCES posts(id) ON DELETE CASCADE
);

-- Table to store replies to comments
CREATE TABLE replies (
    id INT AUTO_INCREMENT PRIMARY KEY,
    comment_id INT NOT NULL,  -- Foreign key for comments
    user_id INT NOT NULL,     -- Assuming you have a users table
    reply TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (comment_id) REFERENCES comments(id) ON DELETE CASCADE
);
